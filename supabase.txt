-- Talisay Online Library Management System - Supabase Schema & Seed Script
-- Safe for repeated execution

-- Enable UUID Extension if not enabled
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. BOOK TYPES TABLE
CREATE TABLE IF NOT EXISTS public.book_types (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL UNIQUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. BOOK SERIES TABLE
CREATE TABLE IF NOT EXISTS public.book_series (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL UNIQUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. BOOKS TABLE
CREATE TABLE IF NOT EXISTS public.books (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    title VARCHAR(255) NOT NULL,
    type_id UUID REFERENCES public.book_types(id) ON DELETE SET NULL,
    series_id UUID REFERENCES public.book_series(id) ON DELETE SET NULL,
    total_copies INT NOT NULL DEFAULT 1 CHECK (total_copies >= 0),
    available_copies INT NOT NULL DEFAULT 1 CHECK (available_copies >= 0),
    status VARCHAR(50) NOT NULL DEFAULT 'Available' CHECK (status IN ('Available', 'Borrowed', 'Reserved', 'Lost', 'Damaged')),
    cover_url VARCHAR(500),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Add cover_url column to existing books tables (safe to re-run)
ALTER TABLE public.books ADD COLUMN IF NOT EXISTS cover_url VARCHAR(500);

-- 3b. BOOK COPIES TABLE (Each physical copy has an auto-generated unique serial number)
CREATE TABLE IF NOT EXISTS public.book_copies (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    book_id UUID NOT NULL REFERENCES public.books(id) ON DELETE CASCADE,
    serial_number VARCHAR(100) NOT NULL UNIQUE,
    status VARCHAR(50) NOT NULL DEFAULT 'Available' CHECK (status IN ('Available', 'Borrowed')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Safe migrations for serial numbers across requests, records, and returns
ALTER TABLE public.borrow_requests ADD COLUMN IF NOT EXISTS serial_number VARCHAR(100);
ALTER TABLE public.borrow_records ADD COLUMN IF NOT EXISTS serial_number VARCHAR(100);
ALTER TABLE public.returns ADD COLUMN IF NOT EXISTS serial_number VARCHAR(100);

-- 4. BORROW REQUESTS TABLE
CREATE TABLE IF NOT EXISTS public.borrow_requests (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    inquiry_number VARCHAR(50) NOT NULL,
    student_name VARCHAR(255) NOT NULL,
    student_id VARCHAR(100) NOT NULL,
    course VARCHAR(100) NOT NULL,
    year_level VARCHAR(50) NOT NULL,
    section VARCHAR(50) NOT NULL,
    book_id UUID NOT NULL REFERENCES public.books(id) ON DELETE CASCADE,
    duration_days INT NOT NULL CHECK (duration_days >= 1 AND duration_days <= 3),
    request_date TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'Pending' CHECK (status IN ('Pending', 'Approved', 'Rejected'))
);

-- 5. BORROW RECORDS TABLE
CREATE TABLE IF NOT EXISTS public.borrow_records (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    request_id UUID REFERENCES public.borrow_requests(id) ON DELETE SET NULL,
    student_name VARCHAR(255) NOT NULL,
    student_id VARCHAR(100) NOT NULL,
    book_id UUID NOT NULL REFERENCES public.books(id) ON DELETE CASCADE,
    borrow_date TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    due_date TIMESTAMP WITH TIME ZONE NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'Borrowed' CHECK (status IN ('Borrowed', 'Returned', 'Overdue'))
);

-- 6. RETURNS TABLE
CREATE TABLE IF NOT EXISTS public.returns (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    record_id UUID NOT NULL REFERENCES public.borrow_records(id) ON DELETE CASCADE,
    student_name VARCHAR(255) NOT NULL,
    book_title VARCHAR(255) NOT NULL,
    borrow_date TIMESTAMP WITH TIME ZONE NOT NULL,
    return_date TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL,
    status VARCHAR(50) NOT NULL DEFAULT 'Returned'
);

-- 7. ADMIN USERS / PROFILES TABLE
CREATE TABLE IF NOT EXISTS public.admin_users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username VARCHAR(100) NOT NULL UNIQUE,
    full_name VARCHAR(255) NOT NULL,
    role VARCHAR(50) NOT NULL DEFAULT 'Admin',
    status VARCHAR(50) NOT NULL DEFAULT 'Active' CHECK (status IN ('Active', 'Disabled')),
    password VARCHAR(255) NOT NULL DEFAULT '',
    plain_password VARCHAR(255) NOT NULL DEFAULT '',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- Add password column to existing admin_users tables (safe to re-run)
ALTER TABLE public.admin_users ADD COLUMN IF NOT EXISTS password VARCHAR(255) NOT NULL DEFAULT '';
ALTER TABLE public.admin_users ADD COLUMN IF NOT EXISTS plain_password VARCHAR(255) NOT NULL DEFAULT '';

-- 8. ACTIVITY LOGS TABLE
CREATE TABLE IF NOT EXISTS public.activity_logs (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    action VARCHAR(255) NOT NULL,
    details TEXT,
    user_name VARCHAR(255) DEFAULT 'System',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- INDEXES
CREATE INDEX IF NOT EXISTS idx_books_type ON public.books(type_id);
CREATE INDEX IF NOT EXISTS idx_books_series ON public.books(series_id);
CREATE INDEX IF NOT EXISTS idx_borrow_requests_student ON public.borrow_requests(student_id);
CREATE INDEX IF NOT EXISTS idx_borrow_requests_status ON public.borrow_requests(status);
CREATE INDEX IF NOT EXISTS idx_borrow_records_status ON public.borrow_records(status);

-- ENABLE ROW LEVEL SECURITY (RLS)
ALTER TABLE public.book_types ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.book_series ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.books ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.borrow_requests ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.borrow_records ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.returns ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.admin_users ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.activity_logs ENABLE ROW LEVEL SECURITY;

-- POLICIES (Allow public reading for books/types/series & submitting requests; authenticated/admin for manage)
DO $$
BEGIN
    -- Public read policy for book types
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Allow public read book_types') THEN
        CREATE POLICY "Allow public read book_types" ON public.book_types FOR SELECT USING (true);
    END IF;
    
    -- Public read policy for book series
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Allow public read book_series') THEN
        CREATE POLICY "Allow public read book_series" ON public.book_series FOR SELECT USING (true);
    END IF;

    -- Public read policy for books
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Allow public read books') THEN
        CREATE POLICY "Allow public read books" ON public.books FOR SELECT USING (true);
    END IF;

    -- Public insert policy for borrow_requests
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Allow public insert borrow_requests') THEN
        CREATE POLICY "Allow public insert borrow_requests" ON public.borrow_requests FOR INSERT WITH CHECK (true);
    END IF;

    -- Admin full control policies (simplified for open access demo/admin mode)
    IF NOT EXISTS (SELECT 1 FROM pg_policies WHERE policyname = 'Allow all access to admin tables') THEN
        CREATE POLICY "Allow all access to admin tables" ON public.admin_users FOR ALL USING (true);
        CREATE POLICY "Allow all access to borrow_requests" ON public.borrow_requests FOR ALL USING (true);
        CREATE POLICY "Allow all access to borrow_records" ON public.borrow_records FOR ALL USING (true);
        CREATE POLICY "Allow all access to returns" ON public.returns FOR ALL USING (true);
        CREATE POLICY "Allow all access to activity_logs" ON public.activity_logs FOR ALL USING (true);
    END IF;
END $$;

-- Open-access policies for the remaining tables (admin UI writes with the anon key)
DROP POLICY IF EXISTS "Allow all access to book_types" ON public.book_types;
CREATE POLICY "Allow all access to book_types" ON public.book_types FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Allow all access to book_series" ON public.book_series;
CREATE POLICY "Allow all access to book_series" ON public.book_series FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Allow all access to books" ON public.books;
CREATE POLICY "Allow all access to books" ON public.books FOR ALL USING (true) WITH CHECK (true);

ALTER TABLE public.book_copies ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS "Allow all access to book_copies" ON public.book_copies;
CREATE POLICY "Allow all access to book_copies" ON public.book_copies FOR ALL USING (true) WITH CHECK (true);
